cmake_minimum_required(VERSION 3.25)
project(HoriSpotAudio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HORI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
# libFLAC 1.5's installed config unconditionally requires Ogg even when built
# WITH_OGG=OFF. Find the public header/library directly for this native-FLAC path.
find_path(HORI_FLAC_INCLUDE FLAC/stream_decoder.h REQUIRED)
find_library(HORI_FLAC_LIBRARY NAMES FLAC REQUIRED)
add_library(hori_flac UNKNOWN IMPORTED)
set_target_properties(hori_flac PROPERTIES IMPORTED_LOCATION "${HORI_FLAC_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${HORI_FLAC_INCLUDE}")
find_library(HORI_OGG_LIBRARY NAMES ogg)
if(HORI_OGG_LIBRARY)
  target_link_libraries(hori_flac INTERFACE "${HORI_OGG_LIBRARY}")
endif()
add_library(hori_audio ../../cspot/audio/FlacDecoder.cpp ../../cspot/src/QualityPolicy.cpp)
target_include_directories(hori_audio PUBLIC ../../cspot/audio ../../cspot/include)
target_link_libraries(hori_audio PUBLIC hori_flac)
target_compile_options(hori_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(APPLE)
  target_sources(hori_audio PRIVATE ../../cspot/audio/RaopPcmReader.cpp)
  target_link_libraries(hori_audio PUBLIC "-framework AudioToolbox" "-framework CoreFoundation")
  add_executable(hori-audio-probe ../../tools/audio-probe.cpp)
  target_link_libraries(hori-audio-probe PRIVATE hori_audio)
endif()
if(HORI_SANITIZE)
  target_compile_options(hori_audio PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hori_audio PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(audio-tests audio-tests.cpp)
target_link_libraries(audio-tests PRIVATE hori_audio)
target_compile_options(audio-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
foreach(case auto lossless vorbis320 legacy96 legacy160 legacy320 strict320 unknown aac24
             decode16 decode24 decode32 mono seek malformed missing corrupt truncated md5)
  add_test(NAME ${case} COMMAND audio-tests ${case} ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
endforeach()
if(APPLE)
  foreach(case pcm16 pcm24 pcm_mono resample48 resample96 seek_resample)
    add_test(NAME ${case} COMMAND audio-tests ${case} ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  endforeach()
endif()
